{\large\bfseries 11th Irish Mathematical Olympiad\\9 May 1998}
\end{center}
\bigskip\noindent\textbf{Paper 1}\bigskip
\begin{enumerate}
\item Show that if $x$ is a nonzero real number, then
\[
x^8 - x^5 - \frac{1}{x} + \frac{1}{x^4} \geq 0.
\]

\item $P$ is a point inside an equilateral triangle such that the distances from $P$ to the three vertices are 3, 4 and 5, respectively. Find the area of the triangle.

\item Show that no integer of the form $xyxy$ in base 10, where $x$ and $y$ are digits, can be the cube of an integer.
Find the smallest base $b > 1$ for which there is a perfect cube of the form $xyxy$ in base $b$.

\item Show that a disc of radius 2 can be covered by seven (possibly overlapping) discs of radius 1.

\item If $x$ is a real number such that $x^2 - x$ is an integer, and, for some $n \geq 3$, $x^n - x$ is also an integer, prove that $x$ is an integer.

\end{enumerate}

\bigskip\noindent\textbf{Paper 2}\bigskip
\begin{enumerate}
\setcounter{enumi}{5}
\item Find all positive integers $n$ that have exactly 16 positive integral divisors $d_1, d_2, \ldots, d_{16}$ such that
\[
1 = d_1 < d_2 < \cdots < d_{16} = n,
\]
$d_6 = 18$ and $d_9 - d_8 = 17$.

\item Prove that if $a$, $b$, $c$ are positive real numbers, then
\begin{enumerate}
\item[(a)]
\[
\frac{9}{a + b + c} \leq 2\left(\frac{1}{a+b} + \frac{1}{b+c} + \frac{1}{c+a}\right),
\]
and
\item[(b)]
\[
\frac{1}{a+b} + \frac{1}{b+c} + \frac{1}{c+a} \leq \frac{1}{2}\left(\frac{1}{a} + \frac{1}{b} + \frac{1}{c}\right).
\]
\end{enumerate}

\item Let $\mathbb{N}$ be the set of all natural numbers (i.e., the positive integers).
\begin{enumerate}
\item[(a)] Prove that $\mathbb{N}$ can be written as a union of three mutually disjoint sets such that, if $m, n \in \mathbb{N}$ and $|m - n| = 2$ or $5$, then $m$ and $n$ are in different sets.
\item[(b)] Prove that $\mathbb{N}$ can be written as a union of four mutually disjoint sets such that, if $m, n \in \mathbb{N}$ and $|m - n| = 2$, $3$ or $5$, then $m$ and $n$ are in different sets. Show, however, that it is impossible to write $\mathbb{N}$ as a union of three mutually disjoint sets with this property.
\end{enumerate}

\item A sequence of real numbers $x_n$ is defined recursively as follows: $x_0, x_1$ are arbitrary positive real numbers, and
\[
x_{n+2} = \frac{1 + x_{n+1}}{x_n}, \quad n = 0, 1, 2, \ldots
\]
Find $x_{1998}$.

\item A triangle $ABC$ has positive integer sides, $\angle A = 2\angle B$ and $\angle C > 90^\circ$. Find the minimum length of its perimeter.

\end{enumerate}
