{\large\bfseries 21st Irish Mathematical Olympiad\\10 May 2008}
\end{center}
\bigskip\noindent\textbf{Paper 1}\bigskip
\begin{enumerate}
\item Let $p_1$, $p_2$, $p_3$ and $p_4$ be four different prime numbers satisfying the equations
\begin{align*}
2p_1 + 3p_2 + 5p_3 + 7p_4 &= 162,\\
11p_1 + 7p_2 + 5p_3 + 4p_4 &= 162.
\end{align*}
Find all possible values of the product $p_1 p_2 p_3 p_4$.

\item For positive real numbers $a$, $b$, $c$ and $d$ such that $a^2 + b^2 + c^2 + d^2 = 1$ prove that
\[
a^2 b^2 cd + ab^2 c^2 d + abc^2 d^2 + a^2 bcd^2 + a^2 bc^2 d + ab^2 cd^2 \;\le\; \frac{3}{32},
\]
and determine the cases of equality.

\item Determine, with proof, all integers $x$ for which $x(x+1)(x+7)(x+8)$ is a perfect square.

\item How many sequences $a_1, a_2, \ldots, a_{2008}$ are there such that each of the numbers $1, 2, \ldots, 2008$ occurs once in the sequence, and $i \in \{a_1, a_2, \ldots, a_i\}$ for each $i$ such that $2 \le i \le 2008$?

\item A triangle $ABC$ has an obtuse angle at $B$. The perpendicular at $B$ to $AB$ meets $AC$ at $D$, and $|CD| = |AB|$. Prove that
\[
|AD|^2 = |AB|\cdot|BC| \quad \text{if and only if} \quad \angle CBD = 30^\circ.
\]
\end{enumerate}

\bigskip\noindent\textbf{Paper 2}\bigskip
\begin{enumerate}
\setcounter{enumi}{5}
\item Find, with proof, all triples of integers $(a,b,c)$ such that $a$, $b$ and $c$ are the lengths of the sides of a right angled triangle whose area is $a + b + c$.

\item Circles $S$ and $T$ intersect at $P$ and $Q$, with $S$ passing through the centre of $T$. Distinct points $A$ and $B$ lie on $S$, inside $T$, and are equidistant from the centre of $T$. The line $PA$ meets $T$ again at $D$. Prove that $|AD| = |PB|$.

\item Find $a_3, a_4, \ldots, a_{2008}$, such that $a_i = \pm 1$ for $i = 3, \ldots, 2008$ and
\[
\sum_{i=3}^{2008} a_i 2^i = 2008,
\]
and show that the numbers $a_3, a_4, \ldots, a_{2008}$ are uniquely determined by these conditions.

\item Given $k \in \{0,1,2,3\}$ and a positive integer $n$, let $f_k(n)$ be the number of sequences $x_1, \ldots, x_n$, where $x_i \in \{-1, 0, 1\}$ for $i = 1, \ldots, n$, and
\[
x_1 + \cdots + x_n \equiv k \pmod{4}.
\]

\begin{enumerate}
\item Prove that $f_1(n) = f_3(n)$ for all positive integers $n$.

\item Prove that
\[
f_0(n) = \frac{3^n + 2 + (-1)^n}{4}
\]
for all positive integers $n$.
\end{enumerate}

\item Suppose that $x$, $y$ and $z$ are positive real numbers such that $xyz \ge 1$.

\begin{enumerate}
\item Prove that
\[
27 \;\le\; (1+x+y)^2 + (1+y+z)^2 + (1+z+x)^2,
\]
with equality if and only if $x = y = z = 1$.

\item Prove that
\[
(1+x+y)^2 + (1+y+z)^2 + (1+z+x)^2 \;\le\; 3(x+y+z)^2,
\]
with equality if and only if $x = y = z = 1$.
\end{enumerate}
\end{enumerate}
