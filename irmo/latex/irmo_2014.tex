\bigskip\noindent\textbf{Paper 1}\bigskip
\begin{enumerate}
\item Given an $8 \times 8$ chess board, in how many ways can we select 56 squares on the board while satisfying both of the following requirements:
\begin{enumerate}
\item[(a)] All black squares are selected.
\item[(b)] Exactly seven squares are selected in each column and in each row.
\end{enumerate}

\item Prove for all integers $N > 1$ that $(N^2)^{2014} - (N^{11})^{106}$ is divisible by $N^6 + N^3 + 1$.

\item In the triangle $ABC$, $D$ is the foot of the altitude from $A$ to $BC$, and $M$ is the midpoint of the line segment $BC$. The three angles $\angle BAD$, $\angle DAM$ and $\angle MAC$ are all equal. Find the angles of the triangle $ABC$.

\item Three different nonzero real numbers $a$, $b$, $c$ satisfy the equations
\[
a + \frac{2}{b} = b + \frac{2}{c} = c + \frac{2}{a} = p
\]
where $p$ is a real number. Prove that $abc + 2p = 0$.

\item Suppose that $a_1, \ldots, a_n > 0$, where $n > 1$ and $\sum_{i=1}^n a_i = 1$. For each $i = 1, 2, \ldots, n$, let $b_i = a_i^2 / \sum_{j=1}^n a_j^2$. Prove that
\[
\sum_{i=1}^{n} \frac{a_i}{1 - a_i} \leq \sum_{i=1}^{n} \frac{b_i}{1 - b_i}.
\]
When does equality occur?
\end{enumerate}

\bigskip\noindent\textbf{Paper 2}\bigskip
\begin{enumerate}
\setcounter{enumi}{5}
\item Each of the four positive integers $N$, $N+1$, $N+2$, $N+3$ has exactly six positive divisors. There are exactly 20 different positive numbers which are exact divisors of at least one of the numbers. One of these is 27. Find all possible values of $N$. (Both 1 and $m$ are counted as divisors of the number $m$.)

\item The square $ABCD$ is inscribed in a circle with centre $O$. Let $E$ be the midpoint of $AD$. The line $CE$ meets the circle again at $F$. The lines $FB$ and $AD$ meet at $H$. Prove $|HD| = 2|AH|$.

\item \begin{enumerate}
\item[(a)] Let $a_0$, $a_1$, $a_2$ be real numbers and consider the polynomial $P(x) = a_0 + a_1 x + a_2 x^2$. Assume that $P(-1)$, $P(0)$ and $P(1)$ are integers. Prove that $P(n)$ is an integer for all integers $n$.
\item[(b)] Let $a_0$, $a_1$, $a_2$, $a_3$ be real numbers and consider the polynomial $Q(x) = a_0 + a_1 x + a_2 x^2 + a_3 x^3$. Assume that there exists an integer $i$ such that $Q(i)$, $Q(i+1)$, $Q(i+2)$ and $Q(i+3)$ are integers. Prove that $Q(n)$ is an integer for all integers $n$.
\end{enumerate}

\item Let $n$ be a positive integer and $a_1, \ldots, a_n$ be positive real numbers. Let $g(x)$ denote the product
\[
(x + a_1) \cdots (x + a_n).
\]
Let $a_0$ be a real number and let
\[
f(x) \;=\; (x - a_0)g(x) \;=\; x^{n+1} + b_1 x^n + b_2 x^{n-1} + \ldots + b_n x + b_{n+1}.
\]
Prove that all the coefficients $b_1, b_2, \ldots, b_{n+1}$ of the polynomial $f(x)$ are negative if and only if
\[
a_0 > a_1 + a_2 + \ldots + a_n.
\]

\item Over a period of $k$ consecutive days, a total of 2014 babies were born in a certain city, with at least one baby being born each day. Show that:
\begin{enumerate}
\item[(a)] If $1014 < k \leq 2014$, there must be a period of consecutive days during which exactly 100 babies were born.
\item[(b)] By contrast, if $k = 1014$, such a period might not exist.
\end{enumerate}
\end{enumerate}
