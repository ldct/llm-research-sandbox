\bigskip\noindent\textbf{Paper 1}\bigskip
\begin{enumerate}
\item Let $x, y$ be positive integers, with $y > 3$, and
\[x^2 + y^4 = 2[(x-6)^2 + (y+1)^2].\]
Prove that $x^2 + y^4 = 1994$.

\item Let $A, B, C$ be three collinear points, with $B$ between $A$ and $C$. Equilateral triangles $ABD$, $BCE$, $CAF$ are constructed with $D, E$ on one side of the line $AC$ and $F$ on the opposite side. Prove that the centroids of the triangles are the vertices of an equilateral triangle. Prove that the centroid of this triangle lies on the line $AC$.

\item Determine, with proof, all real polynomials $f$ satisfying the equation
\[f(x^2) = f(x)f(x-1),\]
for all real numbers $x$.

\item Consider the set of $m \times n$ matrices with every entry either 0 or 1. Determine the number of such matrices with the property that the number of ``1''s in each row and in each column is even.

\item Let $f(n)$ be defined on the set of positive integers by the rules: $f(1) = 2$ and
\[f(n+1) = (f(n))^2 - f(n) + 1, \qquad n = 1, 2, 3, \ldots\]
Prove that, for all integers $n > 1$,
\[1 - \frac{1}{2^{2^{n-1}}} < \frac{1}{f(1)} + \frac{1}{f(2)} + \ldots + \frac{1}{f(n)} < 1 - \frac{1}{2^{2^n}}.\]
\end{enumerate}

\bigskip\noindent\textbf{Paper 2}\bigskip
\begin{enumerate}
\setcounter{enumi}{5}
\item A sequence $x_n$ is defined by the rules: $x_1 = 2$ and
\[nx_n = 2(2n-1)x_{n-1}, \qquad n = 2, 3, \ldots\]
Prove that $x_n$ is an integer for every positive integer $n$.

\item Let $p, q, r$ be distinct real numbers that satisfy the equations
\begin{align*}
q &= p(4-p),\\
r &= q(4-q),\\
p &= r(4-r).
\end{align*}
Find all possible values of $p + q + r$.

\item Prove that, for every integer $n > 1$,
\[n\left((n+1)^{2/n} - 1\right) < \sum_{i=1}^{n} \frac{2i+1}{i^2} < n\left(1 - n^{-2/(n-1)}\right) + 4.\]

\item Let $w, a, b$ and $c$ be \emph{distinct} real numbers with the property that there exist real numbers $x$, $y$ and $z$ for which the following equations hold:
\begin{align*}
x + y + z &= 1,\\
xa^2 + yb^2 + zc^2 &= w^2,\\
xa^3 + yb^3 + zc^3 &= w^3,\\
xa^4 + yb^4 + zc^4 &= w^4.
\end{align*}
Express $w$ in terms of $a$, $b$ and $c$.

\item If a square is partitioned into $n$ convex polygons, determine the maximum number of edges present in the resulting figure.
\end{enumerate}
