\bigskip\noindent\textbf{Paper 1}\bigskip
\begin{enumerate}
\item In a triangle $ABC$, $AB = 20$, $AC = 21$ and $BC = 29$. The points $D$ and $E$ lie on the line segment $BC$, with $BD = 8$ and $EC = 9$. Calculate the angle $\angle DAE$.

\item \begin{enumerate}
\item[(a)] A group of people attends a party. Each person has at most three acquaintances in the group, and if two people do not know each other, then they have a mutual acquaintance in the group. What is the maximum number of people present?
\item[(b)] If, in addition, the group contains three mutual acquaintances (i.e., three people each of whom knows the other two), what is the maximum number of people?
\end{enumerate}

\item Find all triples of positive integers $(p, q, n)$, with $p$ and $q$ primes, satisfying
\[
p(p+3) + q(q+3) = n(n+3).
\]

\item Let the sequence $a_1, a_2, a_3, a_4, \ldots$ be defined by
\[
a_1 = 1, \quad a_2 = 1, \quad a_3 = 1
\]
and
\[
a_{n+1}a_{n-2} - a_n a_{n-1} = 2,
\]
for all $n \geq 3$. Prove that $a_n$ is a positive \textit{integer} for all $n \geq 1$.

\item Let $0 < a, b, c < 1$. Prove that
\[
\frac{a}{1-a} + \frac{b}{1-b} + \frac{c}{1-c} \geq \frac{3\sqrt[3]{abc}}{1 - \sqrt[3]{abc}}.
\]
Determine the case of equality.

\end{enumerate}

\bigskip\noindent\textbf{Paper 2}\bigskip
\begin{enumerate}
\setcounter{enumi}{5}
\item A $3 \times n$ grid is filled as follows: the first row consists of the numbers from 1 to $n$ arranged from left to right in ascending order. The second row is a cyclic shift of the top row. Thus the order goes $i, i+1, \ldots, n-1, n, 1, 2, \ldots, i-1$ for some $i$. The third row has the numbers 1 to $n$ in some order, subject to the rule that in each of the $n$ columns, the sum of the three numbers is the same.
For which values of $n$ is it possible to fill the grid according to the above rules? For an $n$ for which this is possible, determine the number of different ways of filling the grid.

\item Suppose $n$ is a product of four distinct primes $a, b, c, d$ such that
\begin{enumerate}
\item[(a)] $a + c = d$;
\item[(b)] $a(a + b + c + d) = c(d - b)$;
\item[(c)] $1 + bc + d = bd$.
\end{enumerate}
Determine $n$.

\item Denote by $\mathbb{Q}$ the set of rational numbers. Determine all functions $f : \mathbb{Q} \longrightarrow \mathbb{Q}$ such that
\[
f(x + f(y)) = y + f(x), \text{ for all } x, y \in \mathbb{Q}.
\]

\item For each real number $x$, define $\lfloor x \rfloor$ to be the greatest integer less than or equal to $x$.
Let $\alpha = 2 + \sqrt{3}$. Prove that
\[
\alpha^n - \lfloor \alpha^n \rfloor = 1 - \alpha^{-n}, \text{ for } n = 0, 1, 2, \ldots
\]

\item Let $ABC$ be a triangle whose side lengths are all integers, and let $D$ and $E$ be the points at which the incircle of $ABC$ touches $BC$ and $AC$ respectively. If $\bigl||AD|^2 - |BE|^2\bigr| \leq 2$, show that $|AC| = |BC|$.

\end{enumerate}
