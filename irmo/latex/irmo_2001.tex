\end{center}
\bigskip\noindent\textbf{Paper 1}\bigskip
\begin{enumerate}
\item Find, with proof, all solutions of the equation
\[
2^n = a! + b! + c!
\]
in positive integers $a$, $b$, $c$ and $n$. (Here, ! means ``factorial''.)

\item Let $ABC$ be a triangle with sides $BC$, $CA$, $AB$ of lengths $a$, $b$, $c$, respectively. Let $D$, $E$ be the midpoints of the sides $AC$, $AB$, respectively. Prove that $BD$ is perpendicular to $CE$ if, and only if,
\[
b^2 + c^2 = 5a^2.
\]

\item Prove that if an odd prime number $p$ can be expressed in the form $x^5 - y^5$, for some integers $x$, $y$, then
\[
\sqrt{\frac{4p+1}{5}} = \frac{v^2 + 1}{2},
\]
for some odd integer $v$.

\item Prove that
\begin{enumerate}
\item[(a)] $\dfrac{2n}{3n+1} \leq \displaystyle\sum_{k=n+1}^{2n} \frac{1}{k}$; \quad and
\item[(b)] $\displaystyle\sum_{k=n+1}^{2n} \frac{1}{k} \leq \dfrac{3n+1}{4(n+1)}$,
\end{enumerate}
for all positive integers $n$.

\item Let $a$, $b$ be real numbers such that $ab > 0$. Prove that
\[
\sqrt[3]{\frac{a^2 b^2 (a+b)^2}{4}} \leq \frac{a^2 + 10ab + b^2}{12}.
\]
Determine when equality occurs.
Hence, or otherwise, prove for \textit{all} real numbers $a$, $b$ that
\[
\sqrt[3]{\frac{a^2 b^2 (a+b)^2}{4}} \leq \frac{a^2 + ab + b^2}{3}.
\]
Determine the cases of equality.

\end{enumerate}

\bigskip\noindent\textbf{Paper 2}\bigskip
\begin{enumerate}
\setcounter{enumi}{5}
\item Find the least positive integer $a$ such that 2001 divides $55^n + a \cdot 32^n$ for some odd integer $n$.

\item Three hoops are arranged concentrically. Each hoop is threaded with 20 beads, of which 10 are black and 10 are white. On each hoop the positions of the beads are labelled 1 through 20 starting at the bottom and travelling counterclockwise.
We say there is a \textit{match} at position $i$ if all three beads at position $i$ have the same colour. We are free to slide all of the beads around any hoop (but not to unthread and rethread them).
Show that it is possible (by sliding) to find a configuration involving at least 5 matches.

\item Let $ABC$ be an acute angled triangle, and let $D$ be the point on the line $BC$ for which $AD$ is perpendicular to $BC$. Let $P$ be a point on the line segment $AD$. The lines $BP$ and $CP$ intersect $AC$ and $AB$ at $E$ and $F$ respectively. Prove that the line $AD$ bisects the angle $EDF$.

\item Determine, with proof, all non-negative real numbers $x$ for which
\[
\sqrt[3]{13 + \sqrt{x}} + \sqrt[3]{13 - \sqrt{x}}
\]
is an integer.

\item Determine, with proof, all functions $f$ from the set of positive integers to itself which satisfy
\[
f(x + f(y)) = f(x) + y
\]
for all positive integers $x$, $y$.

\end{enumerate}
