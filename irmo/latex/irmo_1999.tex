{\large\bfseries 12th Irish Mathematical Olympiad\\8 May 1999}
\end{center}
\bigskip\noindent\textbf{Paper 1}\bigskip
\begin{enumerate}
\item Find all real values $x$ that satisfy
\[
\frac{x^2}{(x + 1 - \sqrt{x+1})^2} < \frac{x^2 + 3x + 18}{(x+1)^2}.
\]

\item Show that there is a positive number in the Fibonacci sequence that is divisible by 1000.
[The Fibonacci sequence $F_n$ is defined by the conditions:
$F_0 = 0$, $F_1 = 1$, $F_n = F_{n-1} + F_{n-2}$ for $n \geq 2$.
So, the sequence begins $0, 1, 1, 2, 3, 5, 8, 13, \ldots$]

\item Let $D$, $E$ and $F$, respectively, be points on the sides $BC$, $CA$ and $AB$, respectively, of a triangle $ABC$ so that $AD$ is perpendicular to $BC$, $BE$ is the angle-bisector of $\angle B$ and $F$ is the mid-point of $AB$. Prove that $AD$, $BE$ and $CF$ are concurrent if, and only if,
\[
a^2(a - c) = (b^2 - c^2)(a + c),
\]
where $a$, $b$ and $c$ are the lengths of the sides $BC$, $CA$ and $AB$, respectively, of the triangle $ABC$.

\item A square floor consists of 10000 squares ($100$ squares $\times$ $100$ squares --- like a large chessboard) is to be tiled. The only available tiles are rectangular $1 \times 3$ tiles, fitting exactly over three squares of the floor.
\begin{enumerate}
\item[(a)] If a $2 \times 2$ square is removed from the centre of the floor, prove that the remaining part of the floor can be tiled with the available tiles.
\item[(b)] If, instead, a $2 \times 2$ square is removed from a corner of the floor, prove that the remaining part of the floor cannot be tiled with the available tiles.
\end{enumerate}
[There are sufficiently many tiles available. To \textit{tile} the floor --- or a portion thereof --- means to completely cover it with the tiles, each tile covering three squares, and no pair of tiles overlapping. The tiles may not be broken or cut.]

\item Three real numbers $a$, $b$, $c$ with $a < b < c$, are said to be in arithmetic progression if $c - b = b - a$.
Define a sequence $u_n$, $n = 0, 1, 2, 3, \ldots$ as follows: $u_0 = 0$, $u_1 = 1$ and, for each $n \geq 1$, $u_{n+1}$ is the smallest positive integer such that $u_{n+1} > u_n$ and $\{u_0, u_1, \ldots, u_n, u_{n+1}\}$ contains no three elements that are in arithmetic progression.
Find $u_{100}$.

\end{enumerate}

\bigskip\noindent\textbf{Paper 2}\bigskip
\begin{enumerate}
\setcounter{enumi}{5}
\item Solve the system of (simultaneous) equations
\begin{align*}
y^2 &= (x+8)(x^2+2),\\
y^2 &= (8+4x)y + 5x^2 - 16x - 16.
\end{align*}

\item A function $f : \mathbb{N} \to \mathbb{N}$ (where $\mathbb{N}$ denotes the set of positive integers) satisfies
\begin{enumerate}
\item[(a)] $f(ab) = f(a)f(b)$ whenever the greatest common divisor of $a$ and $b$ is 1,
\item[(b)] $f(p + q) = f(p) + f(q)$ for all prime numbers $p$ and $q$.
\end{enumerate}
Prove that $f(2) = 2$, $f(3) = 3$ and $f(1999) = 1999$.

\item Let $a$, $b$, $c$ and $d$ be positive real numbers whose sum is 1. Prove that
\[
\frac{a^2}{a+b} + \frac{b^2}{b+c} + \frac{c^2}{c+d} + \frac{d^2}{d+a} \geq \frac{1}{2},
\]
with equality if, and only if, $a = b = c = d = 1/4$.

\item Find all positive integers $m$ with the property that the fourth power of the number of (positive) divisors of $m$ equals $m$.

\item $ABCDEF$ is a convex (not necessarily regular) hexagon with $AB = BC$, $CD = DE$, $EF = FA$ and
\[
\angle ABC + \angle CDE + \angle EFA = 360^\circ.
\]
Prove that the perpendiculars from $A$, $C$ and $E$ to $FB$, $BD$ and $DF$, respectively, are concurrent.

\end{enumerate}
