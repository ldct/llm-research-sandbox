\bigskip\noindent\textbf{Paper 1}\bigskip
\begin{enumerate}
\item The \textit{runcible} positive integers are defined recursively as follows:
\begin{itemize}
\item 1 and 2 are runcible
\item If $a$ and $b$ are runcible (where $a$ and $b$ are not necessarily distinct) then $2a + 3b$ is runcible.
\end{itemize}
Is 2024 runcible?

\item A non-negative integer $p$ is a \textit{3-choice} if $p = k(k-1)(k-2)/6$ for some positive integer $k$. Let $p$ and $q$ be 3-choices with $p < q$. Show there is an integer $n$ such that $p \le n^2 < q$.

\item Let $\mathbb{Z}_+ = \{1, 2, 3, 4, \ldots\}$ be the set of all positive integers. Determine all functions $f : \mathbb{Z}_+ \mapsto \mathbb{Z}_+$ that satisfy
\[
f(mn) + 1 = f(m) + f(n) \text{ for all positive integers } m \text{ and } n;
\]
\[
f(2024) = 1;
\]
\[
f(n) = 1 \text{ for all positive } n \equiv 22 \pmod{23}.
\]

\item How many 4-digit numbers $ABCD$ are there with the property that $|A - B| = |B - C| = |C - D|$?

Note that the first digit $A$ of a four-digit number cannot be zero.

\item Let $A$, $B$, $C$ be three points on a circle $\Gamma$, and let $L$ denote the midpoint of segment $BC$. The perpendicular bisector of $BC$ intersects the circle $\Gamma$ at two points $M$ and $N$, such that $A$ and $M$ are on different sides of line $BC$. Let $S$ denote the point where the segments $BC$ and $AM$ intersect. Line $NS$ intersects the circumcircle of $\triangle ALM$ at two points $D$ and $E$, with $D$ lying in the interior of the circle $\Gamma$.
\begin{enumerate}
\item[(a)] Prove that $M$ is the circumcentre of $\triangle BCD$.
\item[(b)] Prove that the circumcircles of $\triangle BCD$ and $\triangle ADN$ are tangent at the point $D$.
\end{enumerate}
\end{enumerate}

\bigskip\noindent\textbf{Paper 2}\bigskip
\begin{enumerate}
\setcounter{enumi}{5}
\item Find all positive integers $n$ and $m$ such that
\[
\binom{n}{1} + \binom{n}{3} = 2^m.
\]

\item A game of coins is played as follows: You start with 1 head and 1 tail on a table. At each turn, you can perform any one of the following moves:
\begin{enumerate}
\item[a)] You can turn over all the coins on the table.
\item[b)] You can triple the number of heads and tails at the table.
\item[c)] If there are at least 4 tails on the table, you can turn over 4 tails.
\item[d)] If there are at least 5 tails on the table, you can turn over 3 of the tails and discard 2 of the tails.
\end{enumerate}
Knowing that at the end of the game you have 2024 heads, what are all possible numbers of tails at the end of that game?

\item Let $a$, $b$, $c$ be positive real numbers with $a \le c$ and $b \le c$. Prove that
\[
(a + 10b)(b + 22c)(c + 7a) \ge 2024\, abc.
\]

\item Let $K$, $L$, $M$ denote three points on the sides $BC$, $AB$ and $BC$ of $\triangle ABC$, so that $ALKM$ is a parallelogram. Points $S$ and $T$ are chosen on lines $KL$ and $KM$ respectively, so that the quadrilaterals $AKBS$ and $AKCT$ are both cyclic.

Prove that $MLST$ is cyclic if and only if $K$ is the midpoint of $BC$.

\item Let $\mathbb{Z}_+ = \{1, 2, 3, 4, \ldots\}$ be the set of all positive integers. Find, with proof, all functions $f : \mathbb{Z}_+ \mapsto \mathbb{Z}_+$ with the property that
\[
f(x + f(y) + f(f(z))) = z + f(y) + f(f(x))
\]
for all positive integers $x$, $y$, $z$.
\end{enumerate}
