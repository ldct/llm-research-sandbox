\bigskip\noindent\textbf{Paper 1}\bigskip
\begin{enumerate}
\item Find the smallest positive integer $m$ such that $5m$ is an exact $5^{\text{th}}$ power, $6m$ is an exact $6^{\text{th}}$ power, and $7m$ is an exact $7^{\text{th}}$ power.

\item Prove that
\[
1 - \frac{1}{2012}\left(\frac{1}{2} + \frac{1}{3} + \ldots + \frac{1}{2013}\right) \;\geq\; \frac{1}{2^{2012}\sqrt{2013}}.
\]

\item The altitudes of a triangle $ABC$ are used to form the sides of a second triangle $A_1B_1C_1$. The altitudes of $\triangle A_1B_1C_1$ are then used to form the sides of a third triangle $A_2B_2C_2$. Prove that $\triangle A_2B_2C_2$ is similar to $\triangle ABC$.

\item Each of the 36 squares of a $6 \times 6$ table is to be coloured either Red, Yellow or Blue.
\begin{enumerate}
\item[(a)] No row or column is to contain more than two squares of the same colour.
\item[(b)] In any four squares obtained by intersecting two rows with two columns, no colour is to occur exactly three times.
\end{enumerate}
In how many different ways can the table be coloured if both of these rules are to be respected?

\item $A$, $B$ and $C$ are points on the circumference of a circle with centre $O$. Tangents are drawn to the circumcircles of triangles $OAB$ and $OAC$ at $P$ and $Q$ respectively, where $P$ and $Q$ are diametrically opposite $O$. The two tangents intersect at $K$. The line $CA$ meets the circumcircle of $\triangle OAB$ at $A$ and $X$. Prove that $X$ lies on the line $KO$.
\end{enumerate}

\bigskip\noindent\textbf{Paper 2}\bigskip
\begin{enumerate}
\setcounter{enumi}{5}
\item The three distinct points $B$, $C$, $D$ are collinear with $C$ between $B$ and $D$. Another point $A$ not on the line $BD$ is such that $|AB| = |AC| = |CD|$. Prove that $\angle BAC = 36^\circ$ if and only if
\[
\frac{1}{|CD|} - \frac{1}{|BD|} = \frac{1}{|CD| + |BD|}.
\]

\item Consider the collection of different squares which may be formed by sets of four points chosen from the 12 labelled points in the diagram on the right.

For each possible area such a square may have, determine the number of squares which have this area.

Make sure to explain why your list is complete.

\item Find the smallest positive integer $N$ for which the equation $(x^2-1)(y^2-1) = N$ is satisfied by at least two pairs of integers $(x,y)$ with $1 < x \leq y$.

\item We say that a doubly infinite sequence
\[
\ldots, s_{-2}, s_{-1}, s_0, s_1, s_2, \ldots
\]
is \emph{subaveraging} if $s_n = (s_{n-1} + s_{n+1})/4$ for all integers $n$.
\begin{enumerate}
\item[(a)] Find a subaveraging sequence in which all entries are different from each other. Prove that all entries are indeed distinct.
\item[(b)] Show that if $(s_n)$ is a subaveraging sequence such that there exist distinct integers $m$, $n$ such that $s_m = s_n$, then there are infinitely many pairs of distinct integers $i$, $j$ with $s_i = s_j$.
\end{enumerate}

\item Let $a$, $b$, $c$ be real numbers and let $x = a+b+c$, $y = a^2+b^2+c^2$, $z = a^3+b^3+c^3$ and $S = 2x^3 - 9xy + 9z$.
\begin{enumerate}
\item[(a)] Prove that $S$ is unchanged when $a$, $b$, $c$ are replaced by $a+t$, $b+t$, $c+t$, respectively, for any real number $t$.
\item[(b)] Prove that $(3y - x^2)^3 \geq 2S^2$.
\end{enumerate}
\end{enumerate}
